- name: Play 1 - Provision Droplets
  hosts: localhost
  connection: local
  vars:
    do_token: "dop_v1_6df28fa..."
    ssh_key_id: 55756275
    last_name: "Selk"
  tasks:
    - name: Create the WordPress Droplets
      community.digitalocean.digital_ocean_droplet:
        state: present
        oauth_token: "{{ do_token }}"
        name: "{{ item }}"
        size: s-1vcpu-1gb
        image: ubuntu-24-04-x64
        region: nyc3
        ssh_keys: [ "{{ ssh_key_id }}" ]
        unique_name: yes
        loop:
          - "{{ last_name }}F-2604-WP1"
          - "{{ last_name }}F-2604-WP2"

- name: Play 2 - Install LAMP Stack
  hosts: wordpress_servers
  become: yes
  tasks:
    - name: Install Apache, MySQL, and PHP
      apt:
        name:
          - apache2
          - mysql-server
          - php
          - python3-pymysql
        state: present
    - name: Create MySQL Database
      mysql_db:
        name: Selk_Portfolio
        state: present
    - name: Create MySQL User
      mysql_user:
        name: wselk_admin
        password: "Fullsail1!"
        priv: 'Selk_Portfolio.*:ALL'
        state: present
    - name: Download and Unpack WordPress
      unarchive:
        src: https://wordpress.org/latest.tar.gz
        dest: /var/www/html
        remote_src: yes
        extra_opts: [--strip-components=1]
    - name: Set permissions for Apache
      file:
        path: /var/www/html
        owner: www-data
        group: www-data
        recurse: yes
